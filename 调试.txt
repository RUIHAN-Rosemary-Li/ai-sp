export default async function handler(req, res) {
  // 只允许 POST
  if (req.method !== 'POST') {
    return res.status(405).json({ error: 'Method not allowed' });
  }

  // API Key 从环境变量读，前端永远看不到
  const API_KEY = process.env.API_KEY;
  const API_BASE = process.env.API_BASE || 'https://api.deepseek.com';

  try {
    const response = await fetch(`${API_BASE}/v1/chat/completions`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${API_KEY}`
      },
      body: JSON.stringify({
        model: req.body.model || 'deepseek-chat',
        messages: req.body.messages,
        temperature: 0.7
      })
    });

    const data = await response.json();
    res.status(200).json(data);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
}

const sqlite3 = require('sqlite3').verbose();
const bcrypt = require('bcryptjs');

const db = new sqlite3.Database('./ai-sp.db');

db.serialize(() => {
  // 用户表
  db.run(`CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
  )`);

  // 邀请码表
  db.run(`CREATE TABLE IF NOT EXISTS invite_codes (
    code TEXT PRIMARY KEY,
    created_by INTEGER,
    used_by INTEGER DEFAULT NULL,
    used_at DATETIME DEFAULT NULL,
    max_uses INTEGER DEFAULT 1,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(created_by) REFERENCES users(id)
  )`);

  // 会话表（token）
  db.run(`CREATE TABLE IF NOT EXISTS sessions (
    token TEXT PRIMARY KEY,
    user_id INTEGER NOT NULL,
    expires_at DATETIME NOT NULL,
    FOREIGN KEY(user_id) REFERENCES users(id)
  )`);

  // 口试历史表
  db.run(`CREATE TABLE IF NOT EXISTS exam_history (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    case_type TEXT,
    total_score REAL,
    summary TEXT,
    full_detail TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(user_id) REFERENCES users(id)
  )`);

  // 默认管理员（用户名 admin，密码 admin123，首次登录后改）
  const hash = bcrypt.hashSync('admin123', 10);
  db.run(`INSERT OR IGNORE INTO users (username, password_hash) VALUES (?, ?)`, ['admin', hash], function() {
    const adminId = this.lastID || 1;
    // 生成3个邀请码
    const codes = ['ROSE-2024-A1', 'ROSE-2024-B2', 'ROSE-2024-C3'];
    codes.forEach(code => {
      db.run(`INSERT OR IGNORE INTO invite_codes (code, created_by, max_uses) VALUES (?, ?, 1)`, [code, adminId]);
    });
    console.log('✅ 数据库初始化完成');
    console.log('   管理员: admin / admin123');
    console.log('   邀请码: ROSE-2024-A1, ROSE-2024-B2, ROSE-2024-C3');
    console.log('   请首次登录后立即修改密码！');
  });
});

setTimeout(() => { db.close(); process.exit(0); }, 2000);

app.post('/api/auth/register', (req, res) => {
  const { invite_code, username, password } = req.body;
  if (!invite_code || !username || !password) {
    return res.status(400).json({ error: '缺少参数' });
  }
  if (password.length < 6) {
    return res.status(400).json({ error: '密码至少6位' });
  }

  db.get(`SELECT * FROM invite_codes WHERE code = ?`, [invite_code], (err, row) => {
    if (err) return res.status(500).json({ error: '数据库错误' });
    if (!row) return res.status(400).json({ error: '邀请码无效' });
    if (row.used_by) return res.status(400).json({ error: '邀请码已使用' });

    const hash = bcrypt.hashSync(password, 10);
    db.run(`INSERT INTO users (username, password_hash) VALUES (?, ?)`, [username, hash], function(err) {
      if (err) return res.status(400).json({ error: '用户名已存在' });

      const newUserId = this.lastID;
      // 标记邀请码已用
      db.run(`UPDATE invite_codes SET used_by = ?, used_at = CURRENT_TIMESTAMP WHERE code = ?`, [newUserId, invite_code]);

      // 生成 token
      const token = jwt.sign({ userId: newUserId, username }, JWT_SECRET, { expiresIn: '30d' });
      const expires = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000).toISOString();
      db.run(`INSERT INTO sessions (token, user_id, expires_at) VALUES (?, ?, ?)`, [token, newUserId, expires]);

      res.json({ token, username, userId: newUserId });
    });
  });
});

// 登录
app.post('/api/auth/login', (req, res) => {
  const { username, password } = req.body;

  db.get(`SELECT * FROM users WHERE username = ?`, [username], (err, user) => {
    if (err) return res.status(500).json({ error: '数据库错误' });
    if (!user || !bcrypt.compareSync(password, user.password_hash)) {
      return res.status(401).json({ error: '用户名或密码错误' });
    }

    const token = jwt.sign({ userId: user.id, username: user.username }, JWT_SECRET, { expiresIn: '30d' });
    const expires = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000).toISOString();
    db.run(`INSERT OR REPLACE INTO sessions (token, user_id, expires_at) VALUES (?, ?, ?)`, [token, user.id, expires]);

    res.json({ token, username: user.username, userId: user.id });
  });
});

// 鉴权中间件
function requireAuth(req, res, next) {
  const token = req.headers.authorization?.replace('Bearer ', '');
  if (!token) return res.status(401).json({ error: '未登录' });

  db.get(`SELECT s.*, u.username FROM sessions s JOIN users u ON s.user_id = u.id WHERE s.token = ? AND s.expires_at > CURRENT_TIMESTAMP`, [token], (err, session) => {
    if (err || !session) return res.status(401).json({ error: '登录已过期，请重新登录' });
    req.userId = session.user_id;
    req.username = session.username;
    next();
  });
}
